# 数据结构定义

## 供热系统表 (heating_systems) 

|字段名|数据类型|约束|说明|
|----|----|----|----|
|system_id|INT|PRIMARY KEY|系统唯一ID（自增）|
|system_name|VARCHAR(100)|NOT NULL|供热系统名称（如：城北集中供热）|
|location|VARCHAR(200)|NOT NULL|详细地理位置（如：XX市XX区XX路）|
|heating_mode|VARCHAR(50)|NULL|供热模式（集中/分户/太阳能等）|
|heating_type|UINT32|NOT NULL|供热类型(换热站/中间商)|

```SQL
-- 创建供热系统表
CREATE TABLE heating_systems (
    system_id INT AUTO_INCREMENT PRIMARY KEY,
    system_name VARCHAR(100) NOT NULL,
    location VARCHAR(200) NOT NULL,
    heating_mode VARCHAR(50) NULL,
    heating_type INT NOT NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 创建索引优化查询
CREATE INDEX idx_heating_type ON heating_systems(heating_type);
CREATE INDEX idx_system_name ON heating_systems(system_name);
```

## 上游供热系统指标表(upstream_sys_indicator)

|字段名|数据类型|约束|说明|
|----|----|----|----|
|station_id|INT|NOT NULL|供热站id|
|name|VARCHAR(100)|NOT NULL|供热站名称|
|location|VARCHAR(200)|NOT NULL|供热站地理位置|
|pressure|DECIMAL(6,2)|NOT NULL|供热站压力指标(kPa)|
|flow_rate|DECIMAL(8,2)|NOT NULL|供热站流量指标(m³/h)|
|temperature|DECIMAL(5,2)|NOT NULL|供热站温度指标(℃)|
|status|ENUM|NOT NULL|供热站状态|

```SQL
CREATE TABLE upstream_sys_indicator (
    station_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    location VARCHAR(200) NOT NULL,
    pressure DECIMAL(6,2) NOT NULL,   -- 压力(kPa)
    flow_rate DECIMAL(8,2) NOT NULL,   -- 流量(m³/h)
    temperature DECIMAL(5,2) NOT NULL, -- 温度(℃)
    type ENUM('secondary_pipe', 'upstream_provider') NOT NULL,
    status ENUM('normal', 'warning', 'fault') DEFAULT 'normal',
    last_update TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
```

## 供暖监测数据记录表(heating_monitoring_record)

|字段名|数据类型|约束|说明|
|----|----|----|----|
|record_id|BIGINT|AUTO_INCERMENT|记录id(唯一)|
|resident_id|INT|NOT NULL|关联住户id|
|timestamp|TIMESTAMP|NOT NULL|记录上报时间|
|return_temp|DECIMAL(5,2)|NOT NULL|回水温度(℃)|
|flow_rate|DECIMAL(8,2)|NOT NULL|流量数据(m³/h)|

```SQL
-- 供暖监测数据表（高频采集）
CREATE TABLE heating_monitoring_record (
    record_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    resident_id INT NOT NULL,
    timestamp TIMESTAMP(3) NOT NULL,  -- 毫秒级时间戳
    return_temp DECIMAL(5,2) NOT NULL, -- 回水温度(℃)
    flow_rate DECIMAL(8,2) NOT NULL,   -- 流量(m³/h)
    FOREIGN KEY (resident_id) REFERENCES resident_info(resident_id)
);
```


## 小区信息表(community_info)

|字段名|数据类型|约束|说明|
|----|----|----|----|
|name|VARCHAR(100)|NOT NULL|小区名称|
|location|VARCHAR(200)|NOT NULL|小区位置|
|heating_mode|VARCHAR(50)|NOT NULL|小区供热模式|
|upstream_operator_id|INT||上游数据信息id|
|building_count|INT||楼栋数|
|unit_count|INT||单元总数|
|resident_count|INT||住户总数|

```SQL
CREATE TABLE community_info (
    community_id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    location VARCHAR(200) NOT NULL,
    heating_mode VARCHAR(50) NOT NULL,
    upstream_operator_id INT,
    building_count INT DEFAULT 0,
    unit_count INT DEFAULT 0,
    resident_count INT DEFAULT 0,
    FOREIGN KEY (upstream_operator_id) REFERENCES upstream_operators(operator_id)
);
```


## 住户信息表(resident_info)

|字段名|数据类型|约束|说明|
|----|----|----|----|
|resident_id|INT|NOT NULL|住户id|
|community_id|INT|NOT NULL|所属小区id|
|building_id|INT|NOT NULL|所属楼栋编号|
|uint_id|INT|NOT NULL|所属单元编号|
|room_number|VARCHAR(20)||房间门牌号|
|house_type|VARCHAR(50)||户型|
|area|DECIMAL(8,2)|NOT NULL|建筑面积|
|payment_status|ENUM|NOT NULL|缴费状态|

```SQL
CREATE TABLE resident_info (
    resident_id INT AUTO_INCREMENT PRIMARY KEY,
    community_id INT NOT NULL,
    building_id INT NOT NULL,
    unit_id INT NOT NULL,
    room_number VARCHAR(20) NOT NULL,
    house_type VARCHAR(50) NOT NULL, -- 户型
    area DECIMAL(8,2) NOT NULL,      -- 面积
    payment_status ENUM('paid', 'unpaid', 'partial') DEFAULT 'unpaid',
    last_payment_date DATE,
    FOREIGN KEY (community_id) REFERENCES community_info(community_id)
);
```

## 设备信息表(device_info)

|字段名|数据类型|约束|说明|
|----|----|----|----|
|device_id|INT|AUTO_INCREMENT|设备id|
|resident_id|INT|NOT NULL|设备关联用户id|
|device_coding|VARCHAR(100)||设备编码信息|
|device_type|ENUM|NOT NULL|设备类型(温度监测阀门，楼层控制器，单元控制器)|
|battery_remaining_percent|DECIMAL(3,0)|NOT NULL|电池剩余百分比|
|operation_hours|INT||运行时长|
|status|ENUM||设备状态|
|last_check_date|TIMESTAMP||最后一次检查时间|


```SQL
-- 设备状态表
CREATE TABLE device_info (
    device_id INT AUTO_INCREMENT PRIMARY KEY,
    resident_id INT NOT NULL,
    device_coding VARCHAR(100) NOT NULL,
    device_type ENUM('temperature monitoring valve', 'floor controller', 'unit controller') NOT NULL,
    battery_remaining_percent DECIMAL(3,0) DEFAULT 100, -- 电池电量百分比(%)
    operation_hours INT DEFAULT 0,          -- 运行时长(小时)
    status ENUM('online', 'offline', 'error') DEFAULT 'online',
    last_check_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (resident_id) REFERENCES resident_info(resident_id)
);
```

## 外部环境数据表(external_data)

|字段名|数据类型|约束|说明|
|----|----|----|----|
|data_id|BIGINT|AUTO_INCREMENT|环境数据id|
|community_id|INT||数据对应小区id|
|timestamp|TIMESTAMP||数据记录时间戳|
|

```SQL
-- 外部环境数据表
CREATE TABLE external_data (
    data_id BIGINT AUTO_INCREMENT PRIMARY KEY,
    community_id INT NOT NULL,
    timestamp TIMESTAMP NOT NULL,
    forecast_temp DATE NOT NULL,                -- 预报温度
    outdoor_temp DECIMAL(5,2) NOT NULL,         -- 室外温度(℃)
    outdoor_humidity DECIMAL(5,2) NOT NULL,     -- 室外湿度(%)
    wind_speed DECIMAL(6,2) NOT NULL,           -- 风速(m/s)
    wind_direction DECIMAL(5,2),                -- 风向(度)
    light_intensity DECIMAL(8,2),               -- 光照强度(lux)
    weather_condition VARCHAR(50)               -- 天气状况
    FOREIGN KEY(community_id) REFERENCES community_info(community_id)
);
```
